1. SP_ALDA_CANCEL_ASSIGN
ALTER PROCEDURE [dbo].[SP_ALDA_CANCEL_ASSIGN]
    @usercreate VARCHAR(200), @PENUGASAN_ID BIGINT, @CURRENT_VERSION INT, @cancel_reason VARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @existing_status VARCHAR(20), @existing_ver INT, @contract_no VARCHAR(50), @existing_pic_nik VARCHAR(50),
            @existing_pic_branch VARCHAR(10), @cust_contract_status VARCHAR(50), @cust_amount DECIMAL(18,2);

    BEGIN TRY
        BEGIN TRANSACTION;

        SELECT @contract_no = [CONTRACT_NO], @existing_status = [STATUS], @existing_ver = [ASSIGN_VERSION],
               @existing_pic_nik = [PIC_NIK], @existing_pic_branch = [PIC_BRANCH_ID],
               @cust_contract_status = [CONTRACT_STATUS_SNAPSHOT], @cust_amount = [AMOUNT_TO_BE_PAID]
        FROM [dbo].[ALDA_PENUGASAN] WITH (UPDLOCK)
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID;

        IF @contract_no IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Penugasan tidak ditemukan.' [message]; RETURN;
        END

        IF EXISTS (SELECT 1 FROM [dbo].[ALDA_STATUS_REF] WITH (NOLOCK) WHERE [STATUS_CODE] = @existing_status AND [IS_FINAL] = 1)
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'PENUGASAN FINAL TIDAK DAPAT DIBATALKAN' [message]; RETURN;
        END

        IF @existing_ver <> @CURRENT_VERSION
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Data telah diubah oleh pengguna lain (Race Condition).' [message]; RETURN;
        END

        INSERT INTO [dbo].[ALDA_PENUGASAN_HISTORY] (
            [PENUGASAN_ID], [CONTRACT_NO], [CHANGE_TYPE], [CHANGE_REASON], [CHANGED_AT], [CHANGED_BY],
            [STATUS_BEFORE], [STATUS_AFTER], [ASSIGN_VERSION_BEFORE], [ASSIGN_VERSION_AFTER],
            [PIC_NIK_BEFORE], [PIC_BRANCH_ID_BEFORE], [PIC_NIK_AFTER], [PIC_BRANCH_ID_AFTER],
            [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
        ) VALUES (
            @PENUGASAN_ID, @contract_no, 'CANCEL', @cancel_reason, GETDATE(), @usercreate,
            @existing_status, 'CANCELLED', @existing_ver, @existing_ver + 1,
            @existing_pic_nik, @existing_pic_branch, NULL, NULL, @cust_contract_status, @cust_amount
        );

        UPDATE [dbo].[ALDA_PENUGASAN]
        SET [STATUS] = 'CANCELLED', [ASSIGN_VERSION] = @existing_ver + 1, [NOTES] = @cancel_reason,
            [UPDATED_AT] = GETDATE(), [UPDATED_BY] = @usercreate
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [ASSIGN_VERSION] = @CURRENT_VERSION;

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) [success], 'Penugasan berhasil dibatalkan.' [message];
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) [success], ERROR_MESSAGE() [message];
    END CATCH
END;
GO

3. SP_ALDA_GET_PENUGASAN
ALTER PROCEDURE [dbo].[SP_ALDA_GET_PENUGASAN]
    @BRANCH_ID VARCHAR(10), @NOMOR_KONTRAK VARCHAR(50) = '', @STATUS VARCHAR(20) = '', @PIC_NIK VARCHAR(50) = '',
    @DATE_FROM DATETIME = NULL, @DATE_TO DATETIME = NULL
AS
BEGIN
    SET NOCOUNT ON;

    SELECT V.*, SR.[STATUS_LABEL] AS [ASSIGN_STATUS_LABEL]
    FROM [dbo].[VW_ALDA_PENUGASAN_DETAIL] AS V
    LEFT JOIN [dbo].[ALDA_STATUS_REF] AS SR WITH (NOLOCK) ON SR.[STATUS_CODE] = V.[STATUS]
    WHERE V.[BRANCH_ID] = @BRANCH_ID
      AND (@NOMOR_KONTRAK = '' OR V.[CONTRACT_NO] = @NOMOR_KONTRAK)
      AND (@STATUS = '' OR V.[STATUS] = @STATUS)
      AND (@PIC_NIK = '' OR V.[PIC_NIK] = @PIC_NIK)
      AND (@DATE_FROM IS NULL OR V.[CREATED_AT] >= @DATE_FROM)
      AND (@DATE_TO IS NULL OR V.[CREATED_AT] < DATEADD(DAY, 1, @DATE_TO))
    ORDER BY V.[CREATED_AT] DESC;
END;
GO

6. SP_ALDA_PIC_UPDATE_STATUS
ALTER PROCEDURE [dbo].[SP_ALDA_PIC_UPDATE_STATUS]
    @PENUGASAN_ID BIGINT, @PIC_NIK VARCHAR(50), @NEW_STATUS VARCHAR(20), @CURRENT_VERSION INT
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @contract_no VARCHAR(50), @existing_status VARCHAR(20), @existing_ver INT, @new_version INT;

    BEGIN TRY
        BEGIN TRANSACTION;

        SELECT @contract_no = [CONTRACT_NO], @existing_status = [STATUS], @existing_ver = [ASSIGN_VERSION]
        FROM [dbo].[ALDA_PENUGASAN] WITH (UPDLOCK)
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [PIC_NIK] = @PIC_NIK;

        IF @contract_no IS NULL
        BEGIN
            ROLLBACK TRANSACTION;

        END

            ROLLBACK TRANSACTION;

        END

        IF @existing_ver <> @CURRENT_VERSION
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Data telah diubah oleh pengguna lain (Race Condition).' [message]; RETURN;
        END

        SET @new_version = @existing_ver + 1;

        INSERT INTO [dbo].[ALDA_PENUGASAN_HISTORY] (

        )
        SELECT [PENUGASAN_ID], [CONTRACT_NO], 'STATUS_CHANGE', 'PIC merubah status', GETDATE(), @PIC_NIK,

               [PIC_NIK], [PIC_BRANCH_ID], [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
        FROM [dbo].[ALDA_PENUGASAN]
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID;

        SET [STATUS] = @NEW_STATUS, [ASSIGN_VERSION] = @new_version, [UPDATED_AT] = GETDATE(), [UPDATED_BY] = @PIC_NIK
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [ASSIGN_VERSION] = @CURRENT_VERSION;

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) [success], 'Status berhasil diperbarui.' [message];
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) [success], ERROR_MESSAGE() [message];
    END CATCH
END;
GO

    SELECT V.[PENUGASAN_ID], V.[ASSIGN_VERSION], V.[CONTRACT_NO] AS [NOMOR_KONTRAK], V.[PIC_NIK],

8. SP_ALDA_SUBMIT_ASSIGN
ALTER PROCEDURE [dbo].[SP_ALDA_SUBMIT_ASSIGN]
    @usercreate VARCHAR(200), @pic_nik VARCHAR(200), @nomor_kontrak VARCHAR(50), @notes VARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @submission_id BIGINT, @message VARCHAR(500), @success BIT = 0, @existing_id BIGINT,
            @existing_status VARCHAR(20), @existing_ver INT, @existing_pic_nik VARCHAR(50),
            @existing_pic_branch VARCHAR(10), @cust_branch_id VARCHAR(20), @cust_amount DECIMAL(18,2),
            @cust_contract_status VARCHAR(50);

    BEGIN TRY
        BEGIN TRANSACTION;

        SELECT TOP 1 @cust_branch_id = [BRANCH_ID], @cust_contract_status = [CONTRACT_STATUS], @cust_amount = [AMOUNT_TO_BE_PAID]
        FROM [dbo].[MASTER_ALDA] WITH (NOLOCK)
        WHERE [NOMOR_KONTRAK] = @nomor_kontrak;

        IF @cust_branch_id IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'KONTRAK TIDAK DITEMUKAN' [message], NULL [submission_id]; RETURN;
        END

        IF NOT EXISTS (SELECT 1 FROM [dbo].[MASTER_ALDA_PIC] WITH (NOLOCK) WHERE [NIK] = @pic_nik AND [BRANCH_ID] = @cust_branch_id)
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'PIC TIDAK TERDAFTAR DI CABANG KONTRAK INI' [message], NULL [submission_id]; RETURN;
        END

        SET @submission_id = CAST(REPLACE(REPLACE(REPLACE(REPLACE(CONVERT(VARCHAR(23), GETDATE(), 121), '-', ''), ' ', ''), ':', ''), '.', '') AS BIGINT);
        IF @notes IS NULL OR LTRIM(RTRIM(@notes)) = '' SET @notes = 'Assigned via Web';

        SELECT @existing_id = [PENUGASAN_ID], @existing_status = [STATUS], @existing_ver = ISNULL([ASSIGN_VERSION], 1),
               @existing_pic_nik = [PIC_NIK], @existing_pic_branch = [PIC_BRANCH_ID]
        FROM [dbo].[ALDA_PENUGASAN] WITH (UPDLOCK)
        WHERE [CONTRACT_NO] = @nomor_kontrak;

        IF @existing_id IS NOT NULL
        BEGIN
            IF EXISTS (SELECT 1 FROM [dbo].[ALDA_STATUS_REF] WITH (NOLOCK) WHERE [STATUS_CODE] = @existing_status AND [IS_FINAL] = 1)
            BEGIN
                ROLLBACK TRANSACTION;
                SELECT CAST(0 AS BIT) [success], 'KONTRAK DENGAN STATUS FINAL TIDAK DAPAT DI ASSIGN ULANG' [message], NULL [submission_id]; RETURN;
            END

            INSERT INTO [dbo].[ALDA_PENUGASAN_HISTORY] (
                [PENUGASAN_ID], [CONTRACT_NO], [CHANGE_TYPE], [CHANGE_REASON], [CHANGED_AT], [CHANGED_BY],
                [STATUS_BEFORE], [STATUS_AFTER], [ASSIGN_VERSION_BEFORE], [ASSIGN_VERSION_AFTER],
                [PIC_NIK_BEFORE], [PIC_BRANCH_ID_BEFORE], [PIC_NIK_AFTER], [PIC_BRANCH_ID_AFTER],
                [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
            ) VALUES (
                @existing_id, @nomor_kontrak, 'REASSIGN', @notes, GETDATE(), @usercreate, @existing_status, 'ASSIGNED',
                @existing_ver, @existing_ver + 1, @existing_pic_nik, @existing_pic_branch, @pic_nik, @cust_branch_id,
                @cust_contract_status, @cust_amount
            );

            UPDATE [dbo].[ALDA_PENUGASAN]
            SET [PIC_NIK] = @pic_nik, [PIC_BRANCH_ID] = @cust_branch_id, [SUBMISSION_ID] = @submission_id,
                [STATUS] = 'ASSIGNED', [ASSIGN_VERSION] = @existing_ver + 1, [NOTES] = @notes,
                [CONTRACT_STATUS_SNAPSHOT] = @cust_contract_status, [AMOUNT_TO_BE_PAID] = @cust_amount,
                [UPDATED_AT] = GETDATE(), [UPDATED_BY] = @usercreate
            WHERE [CONTRACT_NO] = @nomor_kontrak;
        END
        ELSE
        BEGIN
            INSERT INTO [dbo].[ALDA_PENUGASAN] (
                [CONTRACT_NO], [SUBMISSION_ID], [STATUS], [ASSIGN_VERSION], [NOTES],
                [PIC_NIK], [PIC_BRANCH_ID], [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID], [CREATED_AT], [CREATED_BY]
            ) VALUES (
                @nomor_kontrak, @submission_id, 'ASSIGNED', 1, @notes, @pic_nik, @cust_branch_id,
                @cust_contract_status, @cust_amount, GETDATE(), @usercreate
            );
        END

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) [success], 'SUCCESS' [message], @submission_id [submission_id];
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) [success], ERROR_MESSAGE() [message], NULL [submission_id];
    END CATCH
END;
GO

	@PIHAK_NON_DEBITUR VARCHAR(50),
    @KEBERADAAN_UNIT VARCHAR(50), 
	@PERNAH_DIKUNJUNGI VARCHAR(50), 
	@PENEMUAN_FRAUD VARCHAR(50),
    @INFORMASI_FRAUD VARCHAR(MAX), 
	@HASIL_KUNJUNGAN VARCHAR(MAX),
	@FOTO_PATH VARCHAR(MAX) -- Parameter baru ditambahkan
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @contract_no VARCHAR(50), @existing_ver INT, @new_version INT;

    IF @BERTEMU_DENGAN = 'Debitur' SET @PIHAK_NON_DEBITUR = 'Bertemu Debitur';
    IF @PENEMUAN_FRAUD = 'Tidak Ada' SET @INFORMASI_FRAUD = 'Tidak Ada Fraud';
    IF @LOKASI_KUNJUNGAN = 'Sesuai' SET @REVISI_ALAMAT_KUNJUNGAN = 'Alamat Kunjungan Valid';

    BEGIN TRY
        BEGIN TRANSACTION;

        SELECT @contract_no = [CONTRACT_NO], @existing_ver = [ASSIGN_VERSION]
        FROM [dbo].[ALDA_PENUGASAN] WITH (UPDLOCK)
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [PIC_NIK] = @PIC_NIK;

        IF @contract_no IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Penugasan tidak ditemukan atau Anda bukan PIC.' [message]; RETURN;
        END

        IF @existing_ver <> @CURRENT_VERSION
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Data telah diubah pengguna lain (Race Condition).' [message]; RETURN;
        END

        SET @new_version = @existing_ver + 1;

        INSERT INTO [dbo].[ALDA_RESURVEY_RESULT] (
            [PENUGASAN_ID], [LOKASI_KUNJUNGAN], [REVISI_ALAMAT_KUNJUNGAN], [BERTEMU_DENGAN],
            [PIHAK_NON_DEBITUR], [KEBERADAAN_UNIT], [PERNAH_DIKUNJUNGI], [PENEMUAN_FRAUD],
            [INFORMASI_FRAUD], [HASIL_KUNJUNGAN], [CREATED_AT], [CREATED_BY], [FOTO_PATH]
        ) VALUES (
            @PENUGASAN_ID, @LOKASI_KUNJUNGAN, @REVISI_ALAMAT_KUNJUNGAN, @BERTEMU_DENGAN,
            @PIHAK_NON_DEBITUR, @KEBERADAAN_UNIT, @PERNAH_DIKUNJUNGI, @PENEMUAN_FRAUD,
            @INFORMASI_FRAUD, @HASIL_KUNJUNGAN, GETDATE(), @PIC_NIK, @FOTO_PATH
        );

        INSERT INTO [dbo].[ALDA_PENUGASAN_HISTORY] (
            [PENUGASAN_ID], [CONTRACT_NO], [CHANGE_TYPE], [CHANGE_REASON], [CHANGED_AT], [CHANGED_BY],
            [STATUS_BEFORE], [STATUS_AFTER], [ASSIGN_VERSION_BEFORE], [ASSIGN_VERSION_AFTER],
            [PIC_NIK_BEFORE], [PIC_BRANCH_ID_BEFORE], [PIC_NIK_AFTER], [PIC_BRANCH_ID_AFTER],
            [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
        )
        SELECT [PENUGASAN_ID], [CONTRACT_NO], 'RESURVEY_SUBMIT', 'Form resurvey disubmit beserta foto', GETDATE(), @PIC_NIK,
               [STATUS], 'COMPLETED', [ASSIGN_VERSION], @new_version, [PIC_NIK], [PIC_BRANCH_ID],
               [PIC_NIK], [PIC_BRANCH_ID], [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
        FROM [dbo].[ALDA_PENUGASAN]
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID;

        UPDATE [dbo].[ALDA_PENUGASAN]
        SET [STATUS] = 'COMPLETED', [ASSIGN_VERSION] = @new_version, [UPDATED_AT] = GETDATE(), [UPDATED_BY] = @PIC_NIK
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [ASSIGN_VERSION] = @CURRENT_VERSION;

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) [success], 'Laporan & Foto berhasil disimpan.' [message];
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) [success], ERROR_MESSAGE() [message];
    END CATCH
END;

10. SP_ALDA_TASKLIST_PENUGASAN
ALTER PROCEDURE [dbo].[SP_ALDA_TASKLIST_PENUGASAN]
    @BRANCH_ID VARCHAR(10), 
	@NOMOR_KONTRAK VARCHAR(50) = '', 
	@CONTRACT_STATUS VARCHAR(20) = '', 
	@PORTFOLIO VARCHAR(20) = ''
AS
BEGIN
    SET NOCOUNT ON;

    SELECT 
		A.[BRANCH_ID], A.[BRANCH_NAME], A.[AREA], A.[PORTFOLIO], A.[NOMOR_KONTRAK], A.[CUSTOMER_NAME],
        A.[GO_LIVE_DATE], A.[TANGGAL_BAYAR_ANGS_TERAKHIR], A.[MERK_KENDARAAN], A.[TYPE_KENDARAAN],
        A.[TAHUN_KENDARAAN], A.[CUSTOMER_PHONE], A.[LEGAL_ADDRESS], A.[CONTRACT_STATUS], A.[AMOUNT_TO_BE_PAID]
    FROM [dbo].[MASTER_ALDA] AS A WITH (NOLOCK)
    WHERE A.[BRANCH_ID] = @BRANCH_ID
      AND (@NOMOR_KONTRAK = '' OR A.[NOMOR_KONTRAK] LIKE '%' + @NOMOR_KONTRAK + '%' ESCAPE '\')
      AND (@CONTRACT_STATUS = '' OR A.[CONTRACT_STATUS] = @CONTRACT_STATUS)
      AND (@PORTFOLIO = '' OR A.[PORTFOLIO] = @PORTFOLIO)
      AND NOT EXISTS (
          SELECT 1 
          FROM [dbo].[ALDA_PENUGASAN] P WITH (NOLOCK)
          WHERE P.[CONTRACT_NO] = A.[NOMOR_KONTRAK]
            AND P.[STATUS] <> 'CANCELLED'
      )
    ORDER BY A.[NOMOR_KONTRAK];
END;
GO

11. SP_ALDA_UPDATE_PIC
ALTER PROCEDURE [dbo].[SP_ALDA_UPDATE_PIC]
    @usercreate VARCHAR(200), @pic_nik_new VARCHAR(50), @PENUGASAN_ID BIGINT, @CURRENT_VERSION INT, @notes VARCHAR(500) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @existing_status VARCHAR(20), @existing_ver INT, @contract_no VARCHAR(50), @existing_pic_nik VARCHAR(50),
            @existing_pic_branch VARCHAR(10), @cust_branch_id VARCHAR(20), @cust_contract_status VARCHAR(50),
            @cust_amount DECIMAL(18,2);

    BEGIN TRY
        BEGIN TRANSACTION;

        SELECT @contract_no = [CONTRACT_NO], @existing_status = [STATUS], @existing_ver = [ASSIGN_VERSION],
               @existing_pic_nik = [PIC_NIK], @existing_pic_branch = [PIC_BRANCH_ID]
        FROM [dbo].[ALDA_PENUGASAN] WITH (UPDLOCK)
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID;

        IF @contract_no IS NULL
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Penugasan tidak ditemukan.' [message]; RETURN;
        END

        IF EXISTS (SELECT 1 FROM [dbo].[ALDA_STATUS_REF] WITH (NOLOCK) WHERE [STATUS_CODE] = @existing_status AND [IS_FINAL] = 1)
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'PENUGASAN DENGAN STATUS FINAL TIDAK DAPAT DIUBAH' [message]; RETURN;
        END

        IF @existing_ver <> @CURRENT_VERSION
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'Data telah diubah oleh pengguna lain (Race Condition).' [message]; RETURN;
        END

        SELECT @cust_branch_id = [BRANCH_ID], @cust_contract_status = [CONTRACT_STATUS], @cust_amount = [AMOUNT_TO_BE_PAID]
        FROM [dbo].[MASTER_ALDA] WITH (NOLOCK)
        WHERE [NOMOR_KONTRAK] = @contract_no;

        IF NOT EXISTS (SELECT 1 FROM [dbo].[MASTER_ALDA_PIC] WITH (NOLOCK) WHERE [NIK] = @pic_nik_new AND [BRANCH_ID] = @cust_branch_id)
        BEGIN
            ROLLBACK TRANSACTION;
            SELECT CAST(0 AS BIT) [success], 'PIC BARU TIDAK TERDAFTAR DI CABANG KONTRAK INI' [message]; RETURN;
        END

        IF @notes IS NULL OR LTRIM(RTRIM(@notes)) = '' SET @notes = 'Perubahan PIC via Web';

        INSERT INTO [dbo].[ALDA_PENUGASAN_HISTORY] (
            [PENUGASAN_ID], [CONTRACT_NO], [CHANGE_TYPE], [CHANGE_REASON], [CHANGED_AT], [CHANGED_BY],
            [STATUS_BEFORE], [STATUS_AFTER], [ASSIGN_VERSION_BEFORE], [ASSIGN_VERSION_AFTER],
            [PIC_NIK_BEFORE], [PIC_BRANCH_ID_BEFORE], [PIC_NIK_AFTER], [PIC_BRANCH_ID_AFTER],
            [CONTRACT_STATUS_SNAPSHOT], [AMOUNT_TO_BE_PAID]
        ) VALUES (
            @PENUGASAN_ID, @contract_no, 'REASSIGN', @notes, GETDATE(), @usercreate, @existing_status, 'ASSIGNED',
            @existing_ver, @existing_ver + 1, @existing_pic_nik, @existing_pic_branch, @pic_nik_new, @cust_branch_id,
            @cust_contract_status, @cust_amount
        );

        UPDATE [dbo].[ALDA_PENUGASAN]
        SET [PIC_NIK] = @pic_nik_new, [PIC_BRANCH_ID] = @cust_branch_id, [STATUS] = 'ASSIGNED',
            [ASSIGN_VERSION] = @existing_ver + 1, [NOTES] = @notes, [CONTRACT_STATUS_SNAPSHOT] = @cust_contract_status,
            [AMOUNT_TO_BE_PAID] = @cust_amount, [UPDATED_AT] = GETDATE(), [UPDATED_BY] = @usercreate
        WHERE [PENUGASAN_ID] = @PENUGASAN_ID AND [ASSIGN_VERSION] = @CURRENT_VERSION;

        COMMIT TRANSACTION;
        SELECT CAST(1 AS BIT) [success], 'PIC berhasil diupdate.' [message];
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        SELECT CAST(0 AS BIT) [success], ERROR_MESSAGE() [message];
    END CATCH
END;
GO

        SELECT -1 [LoginStatus], 'Akun sudah tidak aktif.' [Message], NULL [NIK], NULL [NAMA]; RETURN;
    END

    IF @p_Password = 'user.100'
        SELECT 1 [LoginStatus], 'Login Berhasil.' [Message], @p_NIK [NIK], @v_Nama [NAMA];
    ELSE
        SELECT 0 [LoginStatus], 'Password salah.' [Message], NULL [NIK], NULL [NAMA];
END;
